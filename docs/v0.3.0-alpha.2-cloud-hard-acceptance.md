printf '%s\n' "$SPLIT_OUTPUT" | grep -F 'SESSION CATCHUP DETECTED'
printf '%s\n' "$SPLIT_OUTPUT" | grep -F 'PWF_ALPHA2_OWNED_TAIL_SENTINEL'

printf '%s\n' ROOT_ROOT_OWNED_CATCHUP=PASS SYNTHETIC_CROSS_USER=PASS USERPROMPT_LOCAL_ONLY=PASS
